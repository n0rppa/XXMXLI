🔧 API Examples

// Initialize FULL tracker with custom config const tracker = new XXMXLIFullVisitorTracker({ enableMLClassification: true, enableFraudDetection: true, enableSessionRecording: true, enableHeatmaps: true, googleAnalyticsId: 'GA_TRACKING_ID', webhookUrl: 'https://your-api.com/webhook', fraudDetectionLevel: 'HIGH' }); // Access current session data const session = tracker.getCurrentSession(); console.log('Engagement Score:', session.engagementScore); // Track custom events tracker.trackCustomEvent('demo_interaction', { feature: 'button_click', timestamp: Date.now() }); // Get advanced analytics const stats = tracker.getAdvancedStats(); console.log('Total visitors:', stats.total); console.log('Threat level:', stats.threats);
